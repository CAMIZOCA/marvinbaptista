VR Leaderboard — Score tracking with tie-breaking 
interface LeaderboardEntry {
  playerId: string;
  displayName: string;
  score: number;
  timestamp: number;
}

class VRLeaderboard {
  private entries: LeaderboardEntry[] = [];
  private readonly maxEntries: number;

  constructor(maxEntries = 10) {
    this.maxEntries = maxEntries;
  }

  addScore(playerId: string, displayName: string, score: number): void {
    const entry: LeaderboardEntry = {
      playerId,
      displayName,
      score,
      timestamp: Date.now(),
    };
    this.entries.push(entry);
    this.entries.sort((a, b) =>
      b.score !== a.score ? b.score - a.score : a.timestamp - b.timestamp
    );
    this.entries = this.entries.slice(0, this.maxEntries);
  }

  getTopN(n: number): LeaderboardEntry[] {
    return this.entries.slice(0, n);
  }
}
Player State Machine — FSM for VR avatar control 
type PlayerState = 'idle' | 'moving' | 'jumping' | 'interacting' | 'downed';

interface StateTransition {
  from: PlayerState;
  to: PlayerState;
  guard?: () => boolean;
}

class PlayerStateMachine {
  private current: PlayerState = 'idle';
  private readonly transitions: StateTransition[];
  private onEnterCallbacks = new Map<PlayerState, () => void>();

  constructor(transitions: StateTransition[]) {
    this.transitions = transitions;
  }

  onEnter(state: PlayerState, cb: () => void): void {
    this.onEnterCallbacks.set(state, cb);
  }

  transition(target: PlayerState): boolean {
    const valid = this.transitions.some(
      (t) => t.from === this.current && t.to === target && (!t.guard || t.guard())
    );
    if (!valid) return false;
    this.current = target;
    this.onEnterCallbacks.get(target)?.();
    return true;
  }

  get state(): PlayerState { return this.current; }
}
Event Bus — Typed pub/sub for VR scripts 
type EventMap = Record<string, unknown>;
type Listener<T> = (payload: T) => void;

class TypedEventBus<Events extends EventMap> {
  private listeners = new Map<keyof Events, Set<Listener<unknown>>>();

  on<K extends keyof Events>(event: K, listener: Listener<Events[K]>): () => void {
    if (!this.listeners.has(event)) this.listeners.set(event, new Set());
    this.listeners.get(event)!.add(listener as Listener<unknown>);
    return () => this.off(event, listener);
  }

  off<K extends keyof Events>(event: K, listener: Listener<Events[K]>): void {
    this.listeners.get(event)?.delete(listener as Listener<unknown>);
  }

  emit<K extends keyof Events>(event: K, payload: Events[K]): void {
    this.listeners.get(event)?.forEach((l) => l(payload));
  }
}

// Usage in Horizon Worlds
type GameEvents = {
  playerScored: { playerId: string; points: number };
  roundEnded: { winnerId: string };
};

const bus = new TypedEventBus<GameEvents>();
bus.on('playerScored', ({ playerId, points }) => {
  console.log(`${playerId} scored ${points}`);
});
Object Pool — Pooling for VR performance 
interface Poolable {
  id: string;
  active: boolean;
  reset(): void;
}

class ObjectPool<T extends Poolable> {
  private pool: T[] = [];
  private factory: () => T;

  constructor(factory: () => T, initialSize = 10) {
    this.factory = factory;
    for (let i = 0; i < initialSize; i++) this.pool.push(factory());
  }

  acquire(): T {
    const available = this.pool.find((obj) => !obj.active);
    if (available) { available.active = true; return available; }
    const fresh = this.factory();
    fresh.active = true;
    this.pool.push(fresh);
    return fresh;
  }

  release(obj: T): void {
    obj.active = false;
    obj.reset();
  }

  get activeCount(): number {
    return this.pool.filter((o) => o.active).length;
  }
}
